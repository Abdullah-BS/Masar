      كود ESP32 المرجعي (Arduino)ESP32 reference sketch (Arduino)
      
    // Masar floor screen: reference sketch for ESP32 (Arduino core).
// One ESP32 per screen. Set MY_ID to the screen number shown on the panel (S1 = 1).
// Put your LED matrix drawing code inside drawArrow().
const int  MY_ID         = 1;
const char DEFAULT_ARROW = 'U';     // pre-approved default route for this screen
const unsigned long TIMEOUT_MS = 3000;

char shown = 0;
unsigned long lastFrame = 0;
String line;

void drawArrow(char a) {            // U R D L = arrow, X = closed, O = off
  if (a == shown) return;
  shown = a;
  // TODO: draw `a` on your LED matrix here
  Serial.printf("S%d -> %c\n", MY_ID, a);
}

void setup() {
  Serial.begin(115200);
  drawArrow(DEFAULT_ARROW);
}

void loop() {
  while (Serial.available()) {
    char c = Serial.read();
    if (c == '\n') {                // M,<seq>,<point>,<mode>,<arrows>
      int p = line.lastIndexOf(',');
      if (line.startsWith("M,") && p > 0 && (int)line.length() > p + MY_ID) {
        drawArrow(line[p + MY_ID]);
        lastFrame = millis();
      }
      line = "";
    } else if (c != '\r') {
      line += c;
    }
  }
  if (millis() - lastFrame > TIMEOUT_MS) drawArrow(DEFAULT_ARROW);   // fail-safe
}
